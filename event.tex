	\section{مقدمه}

	هدف رویداد روز کوانتوم این است که صنعت و نظام بانکداری ایران، پیش از رسیدن تهدید کوانتومی، نیاز خود را روشن کند
	و توانمندی‌های داخلی و خارجی را در دو محور رمزنگاری پساکوانتومی و مخابرات کوانتومی ببیند.
	دلیل برگزاری آن در این زمان سه نکته است.

	\subsection*{چرا امروز؟}
	\begin{itemize}[rightmargin=0.5cm, itemsep=5pt]
		\item «\textbf{روز کوانتوم}» \textbf{(\lr{Q-Day}) تاریخ مشخصی ندارد.}
		این اصطلاح به روزی گفته می‌شود که کامپیوتر کوانتومی آن‌قدر قدرتمند شود که رمزنگاری کلید عمومی مانند \lr{RSA} را بشکند.
		پیش‌بینی‌های منتشرشده از ۲۰۲۹ (\lr{Gartner}) تا میانه‌ی دهه‌ی ۲۰۳۰ و پس از آن (\lr{Metaculus} و نظرسنجی \lr{Global Risk Institute}) پراکنده است،
		و \lr{NSA} صراحتاً می‌گوید زمان آن را نمی‌داند \citep{qday-rsa}.
		بنابراین برنامه‌ریزی سازمان‌ها باید بر پایه‌ی زمان لازم برای مهاجرت باشد، نه پیش‌بینی تاریخ شکستن \lr{RSA}.

		\item \textbf{داده‌ی امروز می‌تواند فردا خوانده شود.}
		داده‌ی رمزشده را می‌توان از امروز ضبط کرد و پس از ساخته‌شدن چنین ماشینی رمزگشایی کرد
		(\lr{harvest now, decrypt later}). بنابراین داده‌ای که باید سال‌ها محرمانه بماند، صرف‌نظر از تاریخ دقیق، از همین حالا در معرض خطر است \citep{qday-rsa}.
		این نکته همان چیزی است که محور رمزنگاری پساکوانتومی و محور مخابرات کوانتومی (\lr{QKD}) هر کدام از زاویه‌ی خود به آن پاسخ می‌دهند.

		\item \textbf{دولت‌ها ددلاین مهاجرت گذاشته‌اند.}
		طبق جمع‌بندی \lr{qdayiscoming.com}، \lr{NSA} خواسته همه‌ی سامانه‌های امنیت ملی آمریکا تا ۲۰۳۵ مقاوم به کوانتوم شوند،
		و بر اساس دستور اجرایی \lr{14412} مهم‌ترین سامانه‌های فدرال آمریکا باید تا پایان ۲۰۳۰ تبادل کلید پساکوانتومی داشته باشند \citep{qday-deadlines}.
		این‌ها ددلاین مهاجرت‌اند، نه پیش‌بینی زمان شکسته‌شدن \lr{RSA}.
	\end{itemize}

	مهاجرت یک زیرساخت مخابراتی یا بانکی ماه‌ها و سال‌ها زمان می‌برد، و کسی که دیر شروع کند، در روزی که بحران برسد فرصت جبران ندارد.
	از همین رو گفت‌وگوی صنعت، بانکداری، استارتاپ‌ها و پژوهشگران باید از همین حالا آغاز شود.
	هدف ما ترساندن نیست؛ ترغیب به شروع به‌موقع است، چون شروع به‌موقع بسیار ارزان‌تر از پشیمانی دیرهنگام است.

	\subsection*{مشتری اصلی: بانک مرکزی و نظام بانکداری}
	مشتری اصلی این رویداد بانک مرکزی و نظام بانکداری‌اند. اعتماد به پرداخت، انتقال وجه و هویت دیجیتال در این نظام بر رمزنگاری کلید عمومی بنا شده است،
	و داده‌های مالی باید سال‌ها محرمانه و معتبر بمانند؛ به همین دلیل اهمیت آمادگی کوانتومی برای آن‌ها دوچندان است.

	حساسیت این نظام در برابر اختلال را اتفاق اخیر نشان می‌دهد:
	در خرداد ۱۴۰۵ (ژوئن ۲۰۲۶) حمله‌ی سایبری به زیرساخت ارتباطی مشترک بانک‌های ملی، تجارت، صادرات و توسعه‌ی صادرات،
	خدمات الکترونیکی این بانک‌ها را مختل کرد \citep{aa-banks}.
	چند روز بعد خدمات کارتی بانک‌های ملی، صادرات و تجارت دوباره مختل شد \citep{ary-banks}،
	و بنا بر گزارش‌ها اختلال بیش از دو هفته ادامه یافت \citep{iranintl-banks}.
	این اتفاق شکستن رمزنگاری نبود؛ با این حال نشان می‌دهد اختلال در چند سامانه‌ی به‌هم‌پیوسته چقدر سریع به زندگی روزمره می‌رسد.
	رمزنگاری فعلی بر سختی حل یک مسئله بنا شده است
	(\lr{RSA} بر فاکتورگیری و \lr{ECC} بر لگاریتم گسسته) و در برابر یک کامپیوتر کوانتومی به‌اندازه‌ی کافی قدرتمند بی‌معنا می‌شود؛
	آن‌گاه مسئله چند روز اختلال نیست، از دست رفتن پایه‌ی اعتماد است.
	به همین دلیل رویداد با اعلام نیاز بانک مرکزی آغاز می‌شود و استارتاپ‌ها و پژوهشگران به آن نیاز پاسخ می‌دهند.

	\section{هدف و محورها}

	رویداد روز کوانتوم یک گردهمایی یک‌روزه برای نزدیک‌کردن سه گروه به هم است:
	\textbf{نیازمندان} (بانک مرکزی، بانک‌ها و صنایع بزرگ)، \textbf{توانمندان} (استارتاپ‌ها و شرکت‌های فعال در حوزه‌ی امنیت کوانتومی) و
	\textbf{پژوهشگران} (دانشگاه‌ها و پژوهشگاه‌ها).
	تمرکز رویداد بر کسب‌وکار و صنعت است: چه نیازی وجود دارد، چه راه‌حلی در دسترس است، و چه چیزی تا امروز عملاً اجرا شده است.

	رویداد دو محور دارد:
	\begin{itemize}[rightmargin=0.5cm, itemsep=3pt]
		\item \textbf{رمزنگاری پساکوانتومی} (\lr{PQC}): استانداردهای \lr{NIST}، مهاجرت سازمان‌ها، پیاده‌سازی روی سخت‌افزار و ابزارهای کشف و مدیریت رمزنگاری
		\item \textbf{مخابرات کوانتومی}: توزیع کلید کوانتومی (\lr{QKD})، مولد عدد تصادفی کوانتومی (\lr{QRNG}) و شبکه‌های کوانتوم‌ایمن
	\end{itemize}

	\subsection{قالب ارائه‌ها}
	\begin{itemize}[rightmargin=0.5cm, itemsep=3pt]
		\item \textbf{معرفی توانمندی (پیچ):} استارتاپ‌ها و شرکت‌ها محصول و توانمندی خود را معرفی می‌کنند \tg{پیچ}
		\item \textbf{اعلام نیاز (ریورس پیچ):} بانک مرکزی و صنایع و عمودهای بزرگ‌تر نیازمندی‌های خود را اعلام می‌کنند \tg{نیاز}
		\item \textbf{تجربه‌ی عملیاتی:} مهندسان و اساتید، کار اجرایی، تجربی یا پایلوت آزمایشگاهی خود را ارائه می‌کنند؛ ارائه‌ی مروری یا معرفی مقاله و کتاب مدنظر نیست \tg{پایلوت}
	\end{itemize}

	\section{جامعه‌ی کوانتومی پس از رویداد}

	رویداد یک روز است، اما شبکه‌ی ساخته‌شده در آن باید ماندگار باشد. برای این منظور یک جامعه‌ی باز و فعال از صنعت، بانک‌ها،
	استارتاپ‌ها و پژوهشگران حول دو محور رویداد پیشنهاد می‌شود:
	\begin{itemize}[rightmargin=0.5cm, itemsep=4pt]
		\item \textbf{گفت‌وگوی مستمر:} گروه یا کانال در تلگرام و بله برای بحث تخصصی، و در صورت نیاز فضای \lr{Slack} یا \lr{Discord}
		\item \textbf{حضور حرفه‌ای:} صفحه‌ی \lr{LinkedIn} با انتشار اخبار، مقاله‌ها و دستاوردهای کوانتومی به فارسی و انگلیسی
		\item \textbf{حافظه‌ی رویداد:} ضبط و انتشار ویدئوی سخنرانی‌ها و پنل، و خبرنامه‌ی ماهانه
		\item \textbf{شبکه‌ی اعضا:} بانک افراد و شرکت‌ها (پیوست ب و ج) به‌عنوان هسته‌ی اولیه‌ی جامعه، که پس از رویداد با اعضای جدید تکمیل می‌شود
		\item \textbf{دانش مشترک:} محتوای کارگاه‌ها، تمرین‌ها، واژه‌نامه‌ی فارسی اصطلاحات کوانتومی و فهرست منابع منتخب به‌صورت باز در اختیار اعضا
		\item \textbf{عضویت باز:} مهندسان، پژوهشگران و توسعه‌دهندگان بیرون از سازمان‌های برگزارکننده هم بتوانند بدون هزینه‌ی مستقیم به جامعه بپیوندند
		\item \textbf{دوره‌ی بعد:} دستاوردهای جامعه (مقاله، پروژه، پایلوت) در رویداد بعدی معرفی می‌شود و دایره‌ی جامعه را بزرگ‌تر می‌کند
	\end{itemize}
