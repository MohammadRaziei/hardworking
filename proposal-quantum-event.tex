	\section*{چکیده و چشم‌انداز}
	\addcontentsline{toc}{section}{چکیده و چشم‌انداز}

	فناوری‌های کوانتومی به‌سرعت از آزمایشگاه به صنعت می‌آیند و
	مهاجرت به رمزنگاری پساکوانتومی (PQC) برای اپراتورهای مخابراتی یک الزام عملیاتی است،
	نه یک موضوع پژوهشی دور. در ایران هنوز هیچ سازمان صنعتی‌ای خود را به‌عنوان
	\textbf{مرجع و پیشران حوزه کوانتوم} معرفی نکرده است.
	این سند، به‌جای یک رویداد تک‌باره، یک
	\textbf{برنامه پایدار برندینگ و شاخص‌سازی} پیشنهاد می‌کند تا نام
	«مرکز تحقیق و نوآوری همراه اول» در ذهن جامعه علمی و فناوری کشور
	با «کوانتوم» گره بخورد.

	برنامه از هفت فصل تشکیل شده است که هر کدام مخاطب و خروجی مشخصی دارند
	و با هم یک چرخه کامل می‌سازند: \textbf{دیده شدن} (همایش)، \textbf{مهارت‌آموزی} (هکاتون و کارگاه)،
	\textbf{نهادینه شدن} (آکادمی و دپارتمان)، \textbf{ماندگاری} (جامعه و متن‌باز)،
	\textbf{اعتبار علمی} (کتاب و مقاله) و \textbf{مقیاس‌پذیری محتوا} (تولیدات هوش مصنوعی).

	\section{برگزاری همایش و سخنرانی افراد شاخص داخلی و خارجی}

	همایش \lr{Quantum Tech} ستون اصلی دیده شدن برنامه است.
	در این رویداد، چهره‌های شاخص دانشگاهی داخل کشور و پژوهشگران ایرانی و غیرایرانی
	شاغل در دانشگاه‌های برتر جهان گرد هم می‌آیند و مرکز تحقیق و نوآوری همراه اول
	به‌عنوان میزبان و متولی این گفت‌وگو معرفی می‌شود.

	\subsection{محورهای همایش}
	\begin{itemize}[rightmargin=0.5cm, itemsep=4pt]
		\item مخابرات کوانتومی و شبکه‌های کوانتومی (\lr{QKD} و اینترنت کوانتومی)
		\item رمزنگاری کوانتومی و پساکوانتومی (استانداردهای \lr{NIST PQC})
		\item محاسبات کوانتومی و الگوریتم‌ها
		\item حسگری کوانتومی و کاربردهای صنعتی
	\end{itemize}

	\subsection{سخنرانان}
	فهرست پیشنهادی سخنرانان داخلی و خارجی، همراه با معرفی و دلیل انتخاب هر یک،
	در \textbf{پیوست الف} آمده است. استخر گسترده‌تر افراد شناسایی‌شده (مقیم ایران و خارج از کشور)، که منبع انتخاب سخنران، مدرس و منتور است، در \textbf{پیوست ب} آمده است. ترکیب پیشنهادی شامل اساتید برجسته دانشگاه‌های
	داخل کشور (مانند دانشگاه صنعتی شریف و پژوهشگاه دانش‌های بنیادی) و پژوهشگران
	دانشگاه‌هایی مانند \lr{Caltech}، \lr{Maryland}، \lr{Edinburgh}، \lr{Calgary}،
	\lr{EPFL}، \lr{Queensland}، \lr{FSU} و \lr{Leeds} است. سخنرانان خارجی
	می‌توانند به‌صورت حضوری یا آنلاین مشارکت کنند.

	\section{برگزاری هکاتون کوانتومی توسط همراه اول}

	در این فصل، مرکز تحقیق و نوآوری همراه اول \textbf{خودش} یک هکاتون کوانتومی طراحی و برگزار می‌کند،
	تا نام همراه اول به‌عنوان یکی از متولیان اصلی این حوزه در کشور ثبت شود.

	\subsection{پیشینه: اشتیاق جامعه برای هکاتون کوانتومی}
	هکاتون \lr{PolyhaQ} اخیراً (۱۴ و ۱۵ شهریور ۱۴۰۵) در برج فناوری ابن‌سینا،
	پارک علم و فناوری دانشگاه امیرکبیر برگزار شد و به‌عنوان اولین هکاتون ملی
	متمرکز بر کوانتوم معرفی شده است. برگزاری آن نشان می‌دهد که تقاضای جدی
	برای چنین رویدادهایی در جامعه دانشجویی و پژوهشی کشور وجود دارد و فضا برای
	رویدادهای بعدی باز است.
	در سطح بین‌المللی نیز قالب‌های مشابهی مانند \lr{NYC HAQ} (دانشگاه‌های
	کلمبیا، \lr{CCNY} و \lr{NYU Tandon}) با ترکیب «کارگاه آموزشی پیش از رقابت،
	منتورینگ و جایزه» اجرا شده‌اند.
	هکاتون همراه اول با تمرکز بر \textbf{مسائل واقعی صنعت مخابرات} و
	\textbf{رمزنگاری پساکوانتومی}، مکمل و ادامه این حرکت است و با آن رقابت نمی‌کند.
	
	\subsection{طرح هکاتون همراه اول}
	\begin{itemize}[rightmargin=0.5cm, itemsep=4pt]
		\item \textbf{قالب:} هکاتون 2 تا 3 روزه، ترکیبی از حضوری و آنلاین، با یک هفته کارگاه آموزشی پیش از رقابت برای تازه‌واردها
		\item \textbf{چالش‌های صنعتی:} مسائل واقعی مخابراتی، مانند بهینه‌سازی مسیریابی و جایابی با الگوریتم‌های \lr{QAOA} و \lr{Annealing}، مهاجرت به \lr{PQC}، و طراحی پروتکل‌های امن
		\item \textbf{ابزار:} پلتفرم‌های متن‌باز و شبیه‌سازها (مانند \lr{Qiskit} و \lr{PennyLane}) تا مشارکت‌کنندگان به سخت‌افزار خاصی وابسته نباشند
		\item \textbf{داوری و منتورینگ:} منتورها از تیم همراه اول و دانشگاه‌های همکار؛ داوران از صنعت و دانشگاه
		\item \textbf{جوایز:} جایزه نقدی یا فرصت کارآموزی، و معرفی تیم‌های برتر برای ادامه همکاری با مرکز
		\item \textbf{ماندگاری:} همه پروژه‌های هکاتون در سازمان گیت‌هاب مرکز (فصل 5) منتشر می‌شوند
	\end{itemize}

	\section{همکاری با آکادمی همراه و ایجاد دپارتمان کوانتوم}

	برای اینکه برنامه به یک فعالیت دائمی تبدیل شود، پیشنهاد می‌شود
	آموزش‌های کوانتومی در قالب رسمی «آکادمی همراه» ادغام شود و یک واحد سازمانی
	با عنوان \textbf{دپارتمان کوانتوم} تعریف گردد.

	\begin{itemize}[rightmargin=0.5cm, itemsep=4pt]
		\item \textbf{مسیر آموزشی:} سه سطح مقدماتی، متوسط و پیشرفته با گواهی؛ از «آشنایی با کوانتوم برای مهندسان» تا «\lr{PQC} برای تیم‌های امنیت و شبکه»
		\item \textbf{آموزش داخلی:} ارتقای دانش کارکنان فنی و امنیتی همراه اول، به‌ویژه برای آمادگی مهاجرت به \lr{PQC}
		\item \textbf{دپارتمان کوانتوم:} مسئول کارگاه‌ها، برنامه پژوهشی، ارتباط با دانشگاه‌ها و نگهداری سازمان گیت‌هاب
		\item \textbf{همکاری دانشگاهی:} تفاهم‌نامه برای پروژه‌های دانشجویی، کارآموزی و پایان‌نامه با موضوعات مورد نیاز صنعت
		\item \textbf{ارتباط با سیاست‌گذار:} همکاری با نهادهای مرتبط (مانند معاونت علمی) برای همسویی با برنامه‌های ملی
	\end{itemize}

	فهرست شرکت‌ها و مراکز ایرانی و خارجی فعال در حوزه کوانتوم، برای شناسایی شرکای همکاری، حامیان و منابع محتوا، در \textbf{پیوست ج} آمده است.

	\end{itemize}

	\href{https://github.com/mci-quantum-lab/quantum-awesome}{\lr{quantum-awesome}}
	(فهرست منابع منتخب کوانتوم) اولین مخزن آن است. ایده‌های پیشنهادی برای مخزن‌های بعدی:
	\begin{itemize}[rightmargin=0.5cm, itemsep=3pt]
		\item محتوای کارگاه‌ها و تمرین‌های فارسی (\lr{Qiskit}، \lr{QML}، \lr{PQC})

		\item پروژه‌های برگزیده هکاتون
		\item ابزار شناسایی الگوریتم‌های رمزنگاری در حال استفاده در سیستم‌ها (فهرست رمزنگاری یا \lr{CBOM}) برای برنامه‌ریزی مهاجرت به \lr{PQC}
		\item بستر آزمایشی \lr{TLS} ترکیبی (کلاسیک + پساکوانتومی) برای سنجش اثر بر تأخیر و اندازه بسته‌ها در شبکه
	\end{itemize}

	\subsection{مشارکت در پروژه‌های متن‌باز}
	\textbf{پروژه شاخص: پشتیبانی \lr{PQC} در کتابخانه \lr{httpp}.}
	کتابخانه \href{https://github.com/MohammadRaziei/httpp}{\lr{httpp}}
	(کلاینت و سرور \lr{HTTP} برای \lr{Python} و \lr{C++}) برای ارتباط \lr{HTTPS} بر پایه \lr{mbedTLS} ساخته شده،
	و \lr{mbedTLS} در حال حاضر الگوریتم‌های پساکوانتومی را پوشش نمی‌دهد.
	پیشنهاد، ساخت یک \textbf{فورک} در سازمان \lr{mci-quantum-lab} و افزودن \lr{PQC}
	به‌صورت مشابه با آنچه در اکوسیستم \lr{OpenSSL} وجود دارد (مانند پروژه \lr{Open Quantum Safe}) است.

	\begin{mdframed}[linecolor=mci-gold, linewidth=1.5pt, backgroundcolor=mci-gold!5, roundcorner=5pt]
		\textbf{نکته مهندسی:} در کد رمزنگاری، هوش مصنوعی برای تولید اولیه، مستندسازی و تست مفید است،
		ولی نباید جایگزین پیاده‌سازی‌های مرجع شود. مسیر پیشنهادی، \textbf{ادغام پیاده‌سازی‌های بازبینی‌شده موجود}
		(مانند \lr{liboqs}، \lr{PQClean} یا \lr{mlkem-native}) به‌جای نوشتن از صفر،
		همراه با اجرای بردارهای آزمون رسمی (\lr{KAT}) و بازبینی انسانی تخصصی است.
		این رویکرد هم سریع‌تر است و هم برای انتشار عمومی با آرم همراه اول ایمن‌تر.
	\end{mdframed}

	\textbf{سایر ایده‌های مشارکت متن‌باز:}
	\begin{itemize}[rightmargin=0.5cm, itemsep=3pt]
		\item مشارکت در \lr{liboqs} و \lr{oqs-provider} (افزونه \lr{OpenSSL 3} برای الگوریتم‌های پساکوانتومی)
		\item مشارکت در مستندات و آموزش‌های \lr{Qiskit} و \lr{PennyLane}، از جمله ترجمه فارسی
		\item ساخت و انتشار یک شبیه‌ساز سبک \lr{QKD} برای آموزش و تست
		\item مخزن مرجع فارسی برای مهاجرت به \lr{PQC} در شبکه‌های مخابراتی
	\end{itemize}

	\section{تولیدات هوش مصنوعی در حوزه کوانتوم}

	هوش مصنوعی در این برنامه دو نقش دارد: \textbf{ابزار کار} (دستیارها و عامل‌های هوشمند برای کدنویسی کوانتومی، تمرین و تحلیل)
	و \textbf{ماشین تولید محتوا} (ویدیو، انیمیشن و متن آموزشی در مقیاسی که تیم کوچک به‌تنهایی نمی‌تواند).
	هر دو نقش، فصل‌های قبلی را تقویت می‌کنند: کارگاه‌ها (فصل 3) دستیار آموزشی می‌گیرند، هکاتون (فصل 2) منتور هوشمند،
	جامعه (فصل 5) محتوای روزانه، و همایش (فصل 1) فرصت رونمایی از تولیدات.

	\subsection{هوش مصنوعی عامل‌محور (\lr{Agentic AI}) در کوانتوم}

	\subsubsection*{وضعیت فعلی در جهان}
	نمونه‌های زیر نشان می‌دهند که این ترکیب دیگر نظری نیست (منابع در انتهای فصل):
	\begin{itemize}[rightmargin=0.5cm, itemsep=4pt]
		\item \textbf{مایکروسافت} اعلام کرده در برنامه کوانتومی خود به‌طور فزاینده از عامل‌های هوشمند برای طراحی مواد، خودکارسازی آزمایش‌ها
		و تحلیل داده‌های پژوهشی چندین‌ساله استفاده می‌کند و آن را بخشی طبیعی از جریان کار خود می‌داند [1].
		\item \textbf{پاسکال (\lr{Pasqal}) و \lr{Quantonation}} در پژوهشی مشترک (مرداد 1405) نشان دادند یک عامل هوشمند می‌تواند مقاله، ثبت اختراع
		و درخواست به زبان طبیعی را به آزمایش روی پردازنده‌های کوانتومی اتم خنثی واقعی تبدیل کند.
		نتیجه‌گیری آن‌ها این است که مانع ورود به سخت‌افزار کوانتومی کم می‌شود و جامعه کاربران از چند صد تیم متخصص به دانشمندان و مهندسان رشته‌های دیگر گسترش می‌یابد [2].
		\item \textbf{کدنویسی کوانتومی با کمک هوش مصنوعی:} \lr{Qiskit Code Assistant} در \lr{IBM} مدلی است که برای تولید کد \lr{Qiskit} آموزش دیده، در \lr{VS Code} و
		\lr{JupyterLab} کار می‌کند و با معیار اجرایی \lr{Qiskit HumanEval} (101 مسئله) سنجیده می‌شود [3]. یک سرور \lr{MCP} رسمی نیز در مخزن
		\lr{Qiskit/mcp-servers} برای اتصال این دستیار به عامل‌های هوشمند وجود دارد [4]. پژوهش‌هایی مانند \lr{QSpark} نیز دستیارهای متن‌باز مبتنی بر یادگیری تقویتی برای \lr{Qiskit} پیشنهاد کرده‌اند [5].
		\item \textbf{تحلیل «آیا کوانتوم اینجا کمک می‌کند؟»:} پژوهش \lr{QuantumMind} عاملی پیشنهاد می‌کند که پیش از نوشتن برنامه کوانتومی بررسی می‌کند
		مسئله کلاسیک داده‌شده پیش‌شرط‌های یک الگوریتم کوانتومی را دارد یا نه و چه ادعایی درباره سرعت‌گرفتن قابل دفاع است [6].
		\item \textbf{امنیت:} بنا بر یک تحلیل صنعتی (\lr{QNu Labs}، شهریور 1405)، عامل‌های هوشمند به زیرساخت رمزنگاری پایدار و چرخش کلید ماشینی نیاز دارند و
		مهاجرت به \lr{PQC} باید هم‌زمان با امن‌سازی عامل‌ها برنامه‌ریزی شود؛ فهرست \lr{OWASP Top 10 for Agentic Applications} نیز سطح حمله ویژه این سامانه‌ها را فهرست کرده است [7]. (این منبع یک شرکت فروشنده است و باید با احتیاط خوانده شود.)
		\item \textbf{همکاری‌های دانشگاهی:} \lr{IBM} همکاری‌های پژوهشی خود با \lr{IIT Bombay} و \lr{IISc} را در زمینه هوش مصنوعی عامل‌محور و ارکستریشن محاسبات کوانتومی و \lr{HPC} گسترش داده است [8].
	\end{itemize}

	\subsection{تولید محتوای آموزشی کوانتومی با هوش مصنوعی}

	کمبود محتوای فارسی با کیفیت یکی از موانع اصلی ورود دانشجویان و مهندسان به کوانتوم است.
	تولید ویدیو با هوش مصنوعی هزینه و زمان ساخت انیمیشن‌های مفهومی (درهم‌تنیدگی، پروتکل \lr{BB84}، شبکه‌های کوانتومی، حمله «ذخیره حالا، رمزگشایی بعداً»)
	را بسیار کاهش می‌دهد و به تیم اجازه می‌دهد یک \textbf{سری ویدیویی منظم} منتشر کند.

	\subsubsection*{مسیر تولید (Pipeline)}
	\begin{enumerate}[rightmargin=0.5cm, itemsep=3pt]
		\item \textbf{طرح و فیلمنامه:} متخصص کوانتوم موضوع و نکات علمی را تعیین می‌کند؛ هوش مصنوعی متن پیش‌نویس می‌دهد.
		\item \textbf{بازبینی علمی:} تأیید دقت علمی توسط متخصص (الزامی، پیش از هر تولید تصویری).
		\item \textbf{تولید تصویر و ویدیو:} کلیپ‌های کوتاه (معمولاً 5 تا 15 ثانیه) با مدل‌های تولید ویدیو، همراه با نمودارهای دقیق که با ابزار طراحی یا کد ساخته می‌شوند (فرمول‌ها و مدارها نباید توسط مدل ویدیو «کشیده» شوند).
		\item \textbf{صدا و زیرنویس:} روایت فارسی و زیرنویس دوزبانه؛ بازبینی تلفظ اصطلاحات.
		\item \textbf{مونتاژ و برندینگ:} قالب ثابت با آرم مرکز، رنگ و تیتراژ یکسان.
		\item \textbf{انتشار و پایش:} انتشار در پلتفرم‌ها و بررسی بازخورد و اصلاح خطاهای گزارش‌شده.
	\end{enumerate}

	\subsection{ملاحظات کیفی، حقوقی و برندینگ}
	\begin{itemize}[rightmargin=0.5cm, itemsep=4pt]
		\item \textbf{دقت علمی:} مدل‌های ویدیو برای نمایش علمی دقیق ساخته نشده‌اند؛ هر ویدیو پیش از انتشار توسط متخصص بازبینی می‌شود.
		\item \textbf{برچسب‌گذاری:} یوتیوب از سازندگان می‌خواهد محتوای واقع‌نما که با هوش مصنوعی ساخته یا دستکاری شده را اعلام کنند؛
		استفاده برای نگارش فیلمنامه یا زیرنویس خودکار و محتوای آشکارا غیرواقعی یا انیمیشنی معمولاً از این الزام معاف است [9]. پیشنهاد می‌شود
		همه ویدیوها، فراتر از الزام پلتفرم، با عبارت «تولیدشده با کمک هوش مصنوعی و بازبینی‌شده توسط متخصصان همراه اول» منتشر شوند.
		\item \textbf{چهره و صدا:} از شبیه‌سازی چهره یا صدای افراد واقعی (سخنرانان و اساتید) بدون رضایت کتبی صریح استفاده نمی‌شود.
		\item \textbf{حقوق مالکیت:} شرایط استفاده تجاری هر ابزار (و در برخی پلن‌ها حذف واترمارک) باید پیش از انتخاب بررسی شود.
		\item \textbf{دسترسی و پرداخت:} امکان دسترسی و پرداخت از داخل ایران برای هر ارائه‌دهنده باید پیش از خرید بررسی شود؛ برخی مدل‌ها از طریق آگریگیتورها در دسترس‌ترند.
		\item \textbf{پایداری بازار:} قیمت‌ها و حتی نام مدل‌ها هر چند هفته تغییر می‌کنند؛ تولید باید به یک ارائه‌دهنده وابسته نباشد.
	\end{itemize}
