	\section{مقدمه و اهداف رویداد}
	
	مرکز تحقیق و نوآوری همراه اول در نظر دارد یک رویداد تخصصی در حوزه
	\textbf{فناوری‌های کوانتومی} با محوریت چهار محور \textbf{مخابرات کوانتومی}،
	\textbf{رمزنگاری کوانتومی و پساکوانتومی (PQC)}، \textbf{حسگری کوانتومی} و
	\textbf{محاسبات کوانتومی} با حضور
	\textbf{دکتر کشاورزی} (مسئول مرکز تحقیق و توسعه همراه اول) و
	\textbf{دکتر جان‌نثاری} (معاونت علمی ریاست‌جمهوری) برگزار نماید.
	این رویداد با هدف ایجاد بستری برای تبادل دانش، برندینگ واحد تحقیقات، و معرفی
	دستاوردهای علمی مرکز طراحی شده است.
	
	\subsection{اهداف اصلی}
	
	\begin{itemize}[rightmargin=0.5cm, itemsep=4pt]
		\item برندینگ و معرفی مرکز تحقیق و نوآوری همراه اول در سطح جامعه علمی-فناوری کشور
		\item برگزاری سخنرانی‌های تخصصی با حضور اساتید برجسته دانشگاهی و پژوهشگران داخلی و بین‌المللی
		\item رونمایی رسمی از کتاب ترجمه‌شده توسط تیم.
		\item ایجاد شبکه ارتباطی بین متخصصان صنعت و دانشگاه در حوزه فناوری‌های کوانتومی
		\item آشنایی مخاطبان با الگوریتم‌های مقاوم در برابر کوانتوم و استانداردهای NIST PQC
		\item معرفی آخرین دستاوردهای جهانی در حسگری کوانتومی و محاسبات کوانتومی و کاربردهای صنعتی آن‌ها
	\end{itemize}
